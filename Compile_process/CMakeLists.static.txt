if(NOT CP_SOURCES)
    message(STATUS "Compile_process (static): no .cpp sources found in ${CMAKE_CURRENT_SOURCE_DIR}")
else()
    add_library(compile_process STATIC ${CP_SOURCES})
    target_include_directories(compile_process PUBLIC
        ${CMAKE_SOURCE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}
    )
    if(MSVC)
        target_compile_options(compile_process PRIVATE /W4 /permissive-)
    else()
        target_compile_options(compile_process PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
